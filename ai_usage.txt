**Practice 6 — structs and functions** (3 pts + 1 bonus, deadline Oct 12, course: Languages and Compilers Design)

**Task 1 — Structs (1 pt)**
- `struct Name { Type [mut] field ... }` — fields are const unless marked `mut`; field types are built-ins (`i32`, `i64`, `bool`) or structs declared earlier; no self-containment, at least one field.
- Objects declared like variables: `Point p{10, 20}`, `Circle c{p, 5}`.
- IR: named type (`"Point" = type {i32, i32}`), alloca per object, `gep` by index — semantics finds the index, codegen never looks up by name.

**Task 2 — Functions (1 pt)**
- `fn name := (params) -> type { ... exit value }`; top-level only; can be recursive; params are copies (immutable in body); body sees only its params/locals/functions, not top-level variables.
- Two-pass approach: collect all signatures first, then check/generate bodies.
- IR: each function becomes a module function; `exit` → `ret`, call → `call`; prefix function names to avoid clashing with runtime (e.g. `main`, `printf`).

**Task 3 — Program structure, field access, calls (1 pt)**
- Fixed order: `program ::= { struct } { fn } { statement } exit` — struct after a function or function after a statement is an error at that line.
- New operators: `.` (field read/write, chains allowed on the left of `:=`) and `()` (calls); a call alone is not a statement; still no grouping parens or unary minus.
- Semantics: call result type must match; argument count/types match (`i32` arg may go into an `i64` param); every link of a field chain must exist, and every link must be `mut` for writes.

    def parse_struct(self):
        struct_token = self.eat()
        name = self.expect_kind("identifier", "expected a struct name")
        self.expect_line_end()

        if not self.next_line():
            raise CompileError(
                struct_token.line,
                struct_token.column,
                "expected '{' on its own line after 'struct', "
                "found end of input",
            )

        self.expect_text(
            "{",
            "expected '{' on its own line after 'struct'",
        )
        opening = self.tokens[self.pos - 1]
        fields = []

        while self.next_line():
            if self.at("}"):
                self.eat()
                self.expect_line_end()

                if not fields:
                    raise CompileError(
                        opening.line,
                        opening.column,
                        f"struct '{name.text}' has no fields",
                    )

                return StructNode(
                    line=struct_token.line,
                    column=struct_token.column,
                    name=name.text,
                    fields=fields,
                )

            type_token = self.eat()
            if (
                type_token.kind != "identifier"
                and type_token.text not in {"i32", "i64", "bool"}
            ):
                self.error_here("expected a field type")

            mutable = False
            if self.at("mut"):
                self.eat()
                mutable = True

            field_name = self.expect_kind(
                "identifier",
                "expected a field name",
            )
            self.expect_line_end()
            fields.append(
                FieldNode(
                    field_name.line,
                    field_name.column,
                    field_name.text,
                    type_token.text,
                    mutable,
                )
            )

        raise CompileError(
            opening.line,
            opening.column,
            "'{' is never closed",
        )

            raise CompileError(
                node.line,
                node.column,

                    struct_field.line,
                    struct_field.column,
                    f"unknown type '{struct_field.type_name}' "
                    f"of field '{struct_field.name}'",
                )
            if struct_field.name in field_names:
                raise CompileError(
                    struct_field.line,
                    struct_field.column,
                    f"field '{struct_field.name}' is already declared "
                    f"in struct '{node.name}'",
                )
            field_names.add(struct_field.name)

        self.structs[node.name] = node

    def check_struct_init(self, node, struct_node):
        values = [node.init] + node.extra_inits
        fields = struct_node.fields

        if (
            len(values) == 1
            and isinstance(values[0], VarNode)
            and values[0].type == node.type_name
        ):
            return

        if len(values) != len(fields):
            raise CompileError(
                node.line,
                node.column,
                f"struct '{node.type_name}' needs {len(fields)} values "
                f"in {{}}, got {len(values)}",

        for struct_field, value in zip(fields, values):
            self.check_assignable(
                value,
                struct_field.type_name,
                value,
                f"initialise '{struct_field.name}' of '{node.name}'",
            )

        return None
