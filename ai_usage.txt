You now have two positions:

### 2. Add `next_line()`

Inside `Parser`, immediately after `__init__()`, add:

```python
def next_line(self):
    self.line_pos += 1

    while self.line_pos < len(self.lines):
        self.tokens = [
            token
            for token in self.lines[self.line_pos]
            if token.kind != "endline"
        ]
        self.pos = 0

        if self.tokens:
            return True

        self.line_pos += 1

    self.tokens = []
    self.pos = 0
    return False
```

This moves to the next **nonempty line**, resets the token cursor, and returns `False` when there are no more lines.

### 3. Replace `parse_program()`

Replace the whole method with:

```python
def parse_program(self):
    statements = []
    exit_node = None
    last_line = 1

    while self.next_line():
        last_line = self.tokens[0].line

        if exit_node is not None:
            token = self.tokens[0]
            raise CompileError(
                token.line,
                token.column,
                "no statement is allowed after exit",
            )

        if self.at("exit"):
            exit_node = self.parse_exit()
        else:

    return ProgramNode(1, 1, statements, exit_node)
```

## Next — Parse `if` and multiline blocks

This is the main parser change. **Leave your current `parse_program()` unchanged.**

The rule we’ll use is:

> Parsing an `if` finishes on its final `}` line. The caller then moves to the next line.

### 1. Add two helpers inside `Parser`

Put these after `next_line()`:

```python
def peek_line(self):
    index = self.line_pos + 1

    while index < len(self.lines):
        tokens = [
            token
            for token in self.lines[index]
            if token.kind != "endline"
        ]
        if tokens:
            return tokens
        index += 1

    return None


def expect_line_end(self):
    token = self.peek()
    if token is not None:
        raise CompileError(
            token.line,
            token.column,
            f"unexpected '{token.text}' after the statement",
        )
```

- `peek_line()` looks ahead **without moving**.
- `expect_line_end()` rejects extra tokens on the current line.

### 2. Replace `parse_statement()`

```python
def parse_statement(self):
    token = self.peek()
    if token is None:
        self.error_here("expected a statement")

    if token.text in {"i32", "i64", "bool"}:
        return self.parse_decl()

    if token.text == "if":
        return self.parse_if()

    if token.text == "else":
        raise CompileError(
            token.line,
            token.column,
            "'else' without an 'if'",
        )

    if token.kind == "identifier":
        return self.parse_assign()

    raise CompileError(
        token.line,
        token.column,
        f"cannot start a statement with '{token.text}'",
    )
```

### 3. Add `parse_if()`

Inside `Parser`, add:

```python
def parse_if(self):
    if_token = self.eat()
    condition = self.parse_expr()
    self.expect_line_end()

    if not self.next_line():
        raise CompileError(
            if_token.line,
            if_token.column,
            "expected '{' on its own line after 'if', found end of input",
        )

    self.expect_text(
        "{",
        "expected '{' on its own line after 'if'",
    )
    then_block = self.parse_block()
    else_block = None

    following = self.peek_line()
    if following is not None and following[0].text == "else":
        self.next_line()
        else_token = self.eat()
        self.expect_line_end()

        if not self.next_line():
            raise CompileError(
                else_token.line,
                else_token.column,
                "expected '{' on its own line after 'else', found end of input",
            )

        self.expect_text(
            "{",
            "expected '{' on its own line after 'else'",
        )
        else_block = self.parse_block()

    return IfNode(
        line=if_token.line,
        column=if_token.column,
        condition=condition,
        then_block=then_block,
        else_block=else_block,
    )
```

### 4. Add `parse_block()`

The opening `{` has already been consumed by `parse_if()`.

```python
def parse_block(self):
    opening = self.tokens[self.pos - 1]
    self.expect_line_end()

    statements = []
    exit_node = None

    while self.next_line():
        if self.at("}"):
            self.eat()
            self.expect_line_end()

            if not statements and exit_node is None:
                raise CompileError(
                    opening.line,
                    opening.column,
                    "empty block",
                )

            return BlockNode(
                line=opening.line,
                column=opening.column,
                statements=statements,
                exit_node=exit_node,
            )

        if exit_node is not None:
            token = self.peek()
            raise CompileError(
                token.line,
                token.column,
                "statement after 'exit' in the same block",
            )

        if self.at("exit"):
            exit_node = self.parse_exit()
        else:
            statements.append(self.parse_statement())

        self.expect_line_end()

    raise CompileError(
        opening.line,
        opening.column,
        "'{' is never closed",
    )
```

**Why does nesting work?** When a block encounters another `if`, it calls `parse_if()`. That call consumes the nested blocks and returns on their final `}`. The outer block then continues to its next line.

## Next — Add scope handling

Now we start **Task 2: semantic checking**.

Your checker currently has one dictionary:

```python
self.symbols = {}
```

We’ll replace it with a **stack of dictionaries**:

```text
outer scope       → {"x": outer declaration}
inside a block    → {"x": inner declaration}
```

Lookup starts in the innermost scope and works outward.

### 1. Replace `SemanticChecker.__init__()`

```python
def __init__(self):
    self.scopes = [{}]
```

The first dictionary is the program’s outer scope.

### 2. Add `lookup()`

Inside `SemanticChecker`, add:

```python
def lookup(self, node, name):
    for frame in reversed(self.scopes):
        if name in frame:
            return frame[name]

    raise CompileError(
        node.line,
        node.column,
        f"variable '{name}' is used before its declaration",
    )
```

### 3. Replace `visit_decl()`

```python
def visit_decl(self, node):
    current_scope = self.scopes[-1]

    if node.name in current_scope:
        raise CompileError(
            node.line,
            node.column,
            f"variable '{node.name}' is already declared in this block",
        )

    node.init.accept(self)
    self.check_assignable(
        node.init,
        node.type_name,
        node,
        f"initialise '{node.name}'",
    )

    current_scope[node.name] = node
```

Only the **current scope** is checked for duplicates. An outer declaration with the same name is allowed.

The initializer is checked **before** registering the new declaration. This preserves your existing declaration-order behavior.

### 4. Replace `visit_assign()`

```python
def visit_assign(self, node):
    decl = self.lookup(node, node.name)

    if not decl.mutable:
        raise CompileError(
            node.line,
            node.column,
            f"cannot assign to '{node.name}': it is not mut",
        )

    node.decl = decl
    node.value.accept(self)

    self.check_assignable(
        node.value,
        decl.type_name,
        node,
        f"assign to '{node.name}'",
    )
```

### 5. Replace `visit_var()`

```python
def visit_var(self, node):
    decl = self.lookup(node, node.name)
    node.decl = decl
    node.type = decl.type_name
    return node.type
```

This saves the **exact declaration** on `node.decl`, so code generation won’t need to resolve names itself.

### 6. Add `visit_block()`

```python
def visit_block(self, node):
    self.scopes.append({})

    try:
        for statement in node.statements:
            statement.accept(self)

        if node.exit_node is not None:
            node.exit_node.accept(self)
    finally:
        self.scopes.pop()
```

Each block gets a fresh scope. `finally` removes it even if checking raises an error.

## Next — LLVM generation for `!` and variable slots

We’ll make two small changes inside **`CodeGenerator`**.

### 1. Add `visit_not()`

```python
def visit_not(self, node):
    value = node.operand.accept(self)
    return self.builder.not_(value, name="not.result")
```

Your semantic checker guarantees that `value` is an LLVM `i1`—a one-bit boolean. Flipping that bit turns `true` into `false`, and vice versa.

### 2. Add an entry-block allocation helper

All variable slots must be allocated in **`entry`**, even when their declarations appear inside branches.

Add:

```python
def create_entry_alloca(self, type_name, name):
    current_block = self.builder.block
    entry_block = self.main.entry_basic_block

    allocation_builder = ir.IRBuilder(entry_block)
    allocation_builder.position_at_start(entry_block)

    pointer = allocation_builder.alloca(
        self.llvm_type(type_name),
        name=name,
    )

    self.builder.position_at_end(current_block)
    return pointer
```

This helper:

1. Remembers the current block.
2. Uses a separate builder to insert an allocation at the beginning of `entry`.
3. Repositions your normal builder at the end of the original block.

That last step matters: inserting an instruction with another builder can otherwise leave the normal builder’s insertion position outdated.

### 3. Replace `CodeGenerator.visit_decl()`

```python
def visit_decl(self, node):
    value = node.init.accept(self)
    value = self.coerce(value, node.init.type, node.type_name)

    node.pointer = self.create_entry_alloca(
        node.type_name,
        node.name,
    )

    self.builder.store(value, node.pointer)
```

**Only the allocation moves to `entry`.** The initializer calculation and `store` stay where the declaration executes.

### 1. Add `visit_block()`

```python
def visit_block(self, node):
```

This generates the block’s statements and its optional final `exit`.

A source-language block does **not** automatically create an LLVM basic block. The `if` generator creates the control-flow blocks.

### 2. Add `visit_if()`

```python
def visit_if(self, node):
    condition = node.condition.accept(self)

    then_block = self.main.append_basic_block("then")
    else_block = (
        self.main.append_basic_block("else")
        if node.else_block is not None
        else None
    )
    merge_block = self.main.append_basic_block("merge")

    self.builder.cbranch(
        condition,
        then_block,
        else_block if else_block is not None else merge_block,
    )

    self.builder.position_at_end(then_block)
    node.then_block.accept(self)

    if not self.builder.block.is_terminated:
        self.builder.branch(merge_block)

    if else_block is not None:
        self.builder.position_at_end(else_block)
        node.else_block.accept(self)

        if not self.builder.block.is_terminated:
            self.builder.branch(merge_block)

    self.builder.position_at_end(merge_block)
```

The flow is:

```text
                 condition
                /         \
             then         else
                \         /
                   merge
```

Without an `else`, the false path goes directly to `merge`.

**Important:** we check `self.builder.block.is_terminated`, not `then_block.is_terminated`. A nested `if` may move the builder to another block.

If an arm executes `exit`, your existing `visit_exit()` emits `ret`. We must **not** add a branch after that return.

## Next — Add `check.py`

The assignment requires **`python3 check.py`**. We’ll make a Python test runner that prints a results table, without depending on Bash.

### 1. Create the file

Paste this **whole block into your Ubuntu terminal**, from the repository folder:

```bash
cat > check.py <<'PY'
import shutil
import subprocess
import sys
import tempfile
from pathlib import Path


ROOT = Path(__file__).resolve().parent


def normalized(data):
    return data.replace(b"\r\n", b"\n")


def check_test(source, valid, workdir):
    expected_path = source.with_suffix(".expected")
    if not expected_path.exists():
        return False, "missing .expected file"

    expected = normalized(expected_path.read_bytes())
    output = workdir / "output.ll"
    output.unlink(missing_ok=True)

    compiled = subprocess.run(
        [
            sys.executable,
            str(ROOT / "compiler.py"),
            str(source),
            str(output),
        ],
        cwd=ROOT,
        capture_output=True,
    )

    if valid:
        if compiled.returncode != 0:
            return False, compiled.stderr.decode(errors="replace").strip()

        if not output.exists():
            return False, "compiler produced no IR file"

        executed = subprocess.run(
            ["lli", str(output)],
            capture_output=True,
        )

        if executed.returncode != 0:
            return False, executed.stderr.decode(errors="replace").strip()

        actual = normalized(executed.stdout)
    else:
        if compiled.returncode == 0:
            return False, "invalid program compiled successfully"

        if output.exists():
            return False, "failed compilation left an IR file"

        actual = normalized(compiled.stderr)

    if actual != expected:
        return False, f"expected {expected!r}; got {actual!r}"

    return True, ""


def main():
    if shutil.which("lli") is None:
        print("ERROR: lli was not found on PATH.")
        return 1

    passed = 0
    failed = 0

    print(f"{'RESULT':<8} {'KIND':<8} TEST")
    print("-" * 72)

    with tempfile.TemporaryDirectory(prefix="pr5_check_") as directory:
        workdir = Path(directory)

        for folder, valid in [("ok", True), ("err", False)]:
            sources = sorted((ROOT / "tests" / folder).glob("*.txt"))

            if not sources:
                print(f"FAIL     {folder:<8} no tests found")
                failed += 1

            for source in sources:
                success, detail = check_test(source, valid, workdir)

                result = "PASS" if success else "FAIL"
                kind = "valid" if valid else "invalid"
                print(f"{result:<8} {kind:<8} {source.relative_to(ROOT)}")

                if success:
                    passed += 1
                else:
                    failed += 1
                    print(f"         {detail}")

    print()
    print(f"Passed: {passed}")
    print(f"Failed: {failed}")
    return 0 if failed == 0 else 1


if __name__ == "__main__":
    raise SystemExit(main())
PY
```

- Runs valid programs through `lli`.
- Compares invalid programs’ exact diagnostics.

- Checks that failed compilations leave no IR file.
- Automatically deletes its temporary files.

```bash
program ::= { statement } exit

assignment ::= ident ":=" expression
exit ::= "exit" factor [ NL ]

if ::= "if" expression NL block [ "else" NL block ]
block ::= "{" NL { statement } [ exit ] "}" NL

expression ::= arithmetic [ ( "==" | "!=" ) arithmetic ]
arithmetic ::= term { ( "+" | "-" ) term }
term ::= factor { "*" factor }
factor ::= number | "true" | "false" | ident | "!" factor

# NL means the end of a source line.
# Blank lines are ignored.
# The final source line may end at EOF without a newline.
# A block must contain at least one statement or an exit.

# "{", "}", and "else" must stand alone on their lines.
```
