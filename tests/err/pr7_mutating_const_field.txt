struct Point
{
    i32 mut x
    i32 mut y
    fn d2 := () -> i32
    {
        i32 r{x * x + y * y}
        exit r
    }
    fn shift := (i32 dx) -> i32
    {
        x := x + dx
        exit x
    }
    fn moved := (i32 dx) -> Point
    {
        Point q{x + dx, y}
        exit q
    }
}
struct Box
{
    Point center
}
Point p{3, 4}
Box mut b{p}
i32 r{b.center.shift(1)}
exit r
