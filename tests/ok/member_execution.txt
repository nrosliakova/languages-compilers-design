struct Point
{
    i32 mut x
    i32 mut y

    fn d2 := () -> i32
    {
        i32 r{x * x + y * y}
        exit r
    }

    fn shift := (i32 dx) -> i32
    {
        x := x + dx
        exit x
    }

    fn move := (i32 dx) -> i32
    {
        i32 result{shift(dx)}
        exit result
    }
}
Point mut p{3, 4}
i32 before{p.d2()}
i32 moved{p.move(2)}
i32 after{p.d2()}
i32 result{before + moved + after}
exit result
