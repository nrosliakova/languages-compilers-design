struct Point
{
    i32 mut x
    i32 mut y
    fn d2 := () -> i32
    {
        i32 r{x * x + y * y}
        exit r
    }
    fn shift := (i32 dx) -> i32
    {
        x := x + dx
        exit x
    }
    fn moved := (i32 dx) -> Point
    {
        Point q{x + dx, y}
        exit q
    }
}
struct Maker
{
    i32 mut count
    fn make := () -> Point
    {
        count := count + 1
        Point p{count, 0}
        exit p
    }
}
Maker mut m{0}
print(m.make().moved(1))
i32 x{m.make().x}
i32 r{x + m.count}
exit r
