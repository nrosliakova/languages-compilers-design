struct Point
{
    i32 mut x
    i32 mut y
    fn d2 := () -> i32
    {
        i32 r{x * x + y * y}
        exit r
    }
    fn shift := (i32 dx) -> i32
    {
        x := x + dx
        exit x
    }
    fn moved := (i32 dx) -> Point
    {
        Point q{x + dx, y}
        exit q
    }
}
struct Box
{
    Point p
    bool enabled
    i64 big
}
fn origin := () -> Point
{
    Point p{0, 0}
    exit p
}
Point p{3, 4}
Box b{p, false, 6000000000}
print(origin())
print(p.moved(2))
print(b.p)
print(b)
Point q{p.moved(1)}
Point r{b.p}
print(q)
print(r)
exit 0
