struct Point
{
    i32 mut x
    i32 mut y
    fn d2 := () -> i32
    {
        i32 r{x * x + y * y}
        exit r
    }
    fn shift := (i32 dx) -> i32
    {
        x := x + dx
        exit x
    }
    fn moved := (i32 dx) -> Point
    {
        Point q{x + dx, y}
        exit q
    }
}
struct Box
{
    i32 x
    fn local := () -> i32
    {
        Point mut q{1, 2}
        i32 r{q.shift(3)}
        exit r
    }
}
Box b{0}
i32 r{b.local()}
exit r
